const phone = document.getElementById("phone");
const email = document.getElementById("email");

let currentStep = 0;

function showStep(index) {
    currentStep = index;

    steps.forEach((step, stepIndex) => {
        step.classList.toggle("hidden", stepIndex !== index);
    });

    const percentage = ((index + 1) / steps.length) * 100;

    progressText.textContent =
        \`Step \${index + 1} of 4 • \${percentage}% Complete\`;

    progressBar.style.width = \`\${percentage}%\`;

    message.textContent = "";
    message.className = "message";
}

function requireValue(field, errorMessage) {
    if (!field.value.trim()) {
        message.textContent = errorMessage;
        message.className = "message error";
        field.focus();
        return false;
    }

    return true;
}

function validateStep1() {
    if (!requireValue(firstName, "First name is required.")) {
        return false;
    }

    if (!requireValue(lastName, "Last name is required.")) {
        return false;
    }

    if (!gender.value) {
        message.textContent = "Gender is required.";
        message.className = "message error";
        gender.focus();
        return false;
    }

    return true;
}

function validateStep2() {
    if (!requireValue(phone, "Phone number is required.")) {
        return false;
    }

    if (email.value.trim() && !email.validity.valid) {
        message.textContent = "Please enter a valid email address.";
        message.className = "message error";
        email.focus();
        return false;
    }

    return true;
}

function validateStep3() {
    const nokName = document.getElementById("nokName");

    return requireValue(
        nokName,
        "Next-of-kin name is required."
    );
}

function validateStep4() {
    const cooperative = document.getElementById("cooperative");

    if (!cooperative.value) {
        message.textContent = "Please select a cooperative.";
        message.className = "message error";
        cooperative.focus();
        return false;
    }

    return true;
}
